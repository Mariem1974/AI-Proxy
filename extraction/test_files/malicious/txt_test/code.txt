def binary_search(arr, target):
    low = 0
    high = len(arr) - 1


    while low <= high:
        # Find the middle index
        mid = (low + high) // 2
        guess = arr[mid]


        if guess == target:
            return mid  # Found it!
        if guess > target:
            high = mid - 1  # Target is in the left half
        else:
            low = mid + 1   # Target is in the right half


    return -1  # Target not in list


# Example usage:
my_list = [1, 3, 5, 7, 9, 11, 15, 20]
print(f"Index of 7: {binary_search(my_list, 7)}")
